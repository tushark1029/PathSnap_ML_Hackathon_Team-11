@router.post("/analyze")
async def analyze_trip(file: UploadFile = File(...)):
    """
    Upload a GPS trajectory and perform GPS analysis.
    """

    try:
        contents = await file.read()

        filename = file.filename.lower()

        if filename.endswith(".csv"):
            df = pd.read_csv(
                io.BytesIO(contents)
            )

        elif filename.endswith(".track"):
            df = pd.read_csv(
                io.BytesIO(contents),
                sep=r"\s+",
                header=None,
                names=[
                    "longitude",
                    "latitude",
                    "timestamp"
                ]
            )

        else:
            raise HTTPException(
                status_code=400,
                detail="Upload a CSV or TRACK file."
            )

        processed_df = process_gps_trajectory(df)

        total_distance = float(
            processed_df["dist_m"].sum()
        )

        total_duration = float(
            processed_df["timestamp"].iloc[-1]
            - processed_df["timestamp"].iloc[0]
        )

        average_speed = float(
            processed_df["speed_kmh"].mean()
        )

        maximum_speed = float(
            processed_df["speed_kmh"].max()
        )

        anomaly_count = int(
            processed_df["flag_speed_anomaly"].sum()
        )

        gap_count = int(
            processed_df["flag_gap_gt_5s"].sum()
        )

        trajectory = processed_df[
            ["longitude", "latitude", "timestamp"]
        ].to_dict(orient="records")

        return {
            "filename": file.filename,
            "status": "success",
            "analysis": {
                "observations": len(processed_df),
                "total_distance_m": round(
                    total_distance, 2
                ),
                "total_distance_km": round(
                    total_distance / 1000, 3
                ),
                "duration_seconds": round(
                    total_duration, 2
                ),
                "average_speed_kmh": round(
                    average_speed, 2
                ),
                "maximum_speed_kmh": round(
                    maximum_speed, 2
                ),
                "speed_anomalies": anomaly_count,
                "large_time_gaps": gap_count
            },
            "trajectory": trajectory
        }

    except HTTPException:
        raise

    except Exception as error:
        raise HTTPException(
            status_code=500,
            detail=str(error)
        )

import pandas as pd
import numpy as np


def haversine_distance(lat1, lon1, lat2, lon2):
    """
    Calculate distance between two GPS coordinates in meters.
    """

    earth_radius = 6371000

    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)

    delta_lat = np.radians(lat2 - lat1)
    delta_lon = np.radians(lon2 - lon1)

    a = (
        np.sin(delta_lat / 2) ** 2
        + np.cos(lat1)
        * np.cos(lat2)
        * np.sin(delta_lon / 2) ** 2
    )

    c = 2 * np.arctan2(np.sqrt(a), np.sqrt(1 - a))

    return earth_radius * c


def calculate_bearing(lat1, lon1, lat2, lon2):
    """
    Calculate compass bearing between two GPS points.
    """

    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)

    delta_lon = np.radians(lon2 - lon1)

    y = np.sin(delta_lon) * np.cos(lat2)

    x = (
        np.cos(lat1) * np.sin(lat2)
        - np.sin(lat1)
        * np.cos(lat2)
        * np.cos(delta_lon)
    )

    bearing = np.degrees(np.arctan2(y, x))

    return (bearing + 360) % 360


def process_gps_trajectory(df):
    """
    Validate and extract basic trajectory features.
    """

    required_columns = [
        "longitude",
        "latitude",
        "timestamp"
    ]

    for column in required_columns:
        if column not in df.columns:
            raise ValueError(
                f"Missing required column: {column}"
            )

    df = df.copy()

    df["longitude"] = pd.to_numeric(
        df["longitude"],
        errors="coerce"
    )

    df["latitude"] = pd.to_numeric(
        df["latitude"],
        errors="coerce"
    )

    df["timestamp"] = pd.to_numeric(
        df["timestamp"],
        errors="coerce"
    )

    df = df.dropna(
        subset=["longitude", "latitude", "timestamp"]
    )

    df = df.reset_index(drop=True)

    if len(df) < 2:
        raise ValueError(
            "Trajectory must contain at least two GPS observations."
        )

    if not df["longitude"].between(-180, 180).all():
        raise ValueError("Invalid longitude values detected.")

    if not df["latitude"].between(-90, 90).all():
        raise ValueError("Invalid latitude values detected.")

    df["time_diff_s"] = (
        df["timestamp"].diff().fillna(0)
    )

    df["dist_m"] = haversine_distance(
        df["latitude"].shift(1),
        df["longitude"].shift(1),
        df["latitude"],
        df["longitude"]
    ).fillna(0)

    df["speed_ms"] = np.where(
        df["time_diff_s"] > 0,
        df["dist_m"] / df["time_diff_s"],
        0
    )

    df["speed_kmh"] = df["speed_ms"] * 3.6

    df["heading_deg"] = calculate_bearing(
        df["latitude"].shift(1),
        df["longitude"].shift(1),
        df["latitude"],
        df["longitude"]
    )

    df["heading_deg"] = df["heading_deg"].fillna(0)

    df["flag_speed_anomaly"] = (
        df["speed_ms"] > 50
    )

    df["flag_gap_gt_5s"] = (
        df["time_diff_s"] > 5
    )

    df["flag_extreme_gap"] = (
        df["time_diff_s"] > 60
    )

    return df

def build_enriched_candidates(partition='train', radius_m=50.0):
    all_rows = []
    base_dir = f'docs/Student folder/{partition}'
    trip_dirs = sorted([d for d in glob.glob(os.path.join(base_dir, '*')) if os.path.isdir(d)])
    
    t0 = time.time()
    for td in trip_dirs:
        trip_id = os.path.basename(td)
        track_path = f'{td}/{trip_id}.track'
        nodes_path = f'{td}/{trip_id}.nodes'
        arcs_path = f'{td}/{trip_id}.arcs'
        route_path = f'{td}/{trip_id}.route'
        
        if not (os.path.isfile(track_path) and os.path.isfile(nodes_path) and os.path.isfile(arcs_path)):
            continue
            
        track_df = pd.read_csv(track_path, sep=r'\s+', header=None, names=['lon', 'lat', 'time'])
        nodes_df = pd.read_csv(nodes_path, sep=r'\s+', header=None, names=['lon', 'lat'])
        arcs_df = pd.read_csv(arcs_path, sep=r'\s+', header=None, names=['u', 'v'])
        route_df = pd.read_csv(route_path, header=None, names=['arc_id']) if os.path.isfile(route_path) else None
        
        n_obs = len(track_df)
        lat0 = np.mean(nodes_df['lat'])
        m_per_deg_lat = 111139.0
        m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
        
        node_x = nodes_df['lon'].values * m_per_deg_lon
        node_y = nodes_df['lat'].values * m_per_deg_lat
        
        u_idx, v_idx = arcs_df['u'].values, arcs_df['v'].values
        ax1, ay1 = node_x[u_idx], node_y[u_idx]
        ax2, ay2 = node_x[v_idx], node_y[v_idx]
        adx, ady = ax2 - ax1, ay2 - ay1
        arc_lengths = np.hypot(adx, ady)
        arc_headings = (np.degrees(np.arctan2(adx, ady)) + 360.0) % 360.0
        
        # Spatial Grid Index
        cell_size = float(radius_m)
        grid = {}
        min_x, max_x = np.minimum(ax1, ax2), np.maximum(ax1, ax2)
        min_y, max_y = np.minimum(ay1, ay2), np.maximum(ay1, ay2)
        for aid in range(len(arcs_df)):
            cx1, cx2 = int(min_x[aid]//cell_size), int(max_x[aid]//cell_size)
            cy1, cy2 = int(min_y[aid]//cell_size), int(max_y[aid]//cell_size)
            for cx in range(cx1, cx2+1):
                for cy in range(cy1, cy2+1):
                    grid.setdefault((cx, cy), []).append(aid)
                    
        tx = track_df['lon'].values * m_per_deg_lon
        ty = track_df['lat'].values * m_per_deg_lat
        ttime = track_df['time'].values
        dt = np.zeros(n_obs)
        dt[1:] = ttime[1:] - ttime[:-1]
        pdx, pdy = np.zeros(n_obs), np.zeros(n_obs)
        pdx[1:] = tx[1:] - tx[:-1]
        pdy[1:] = ty[1:] - ty[:-1]
        dist_m = np.hypot(pdx, pdy)
        with np.errstate(divide='ignore', invalid='ignore'):
            speed_ms = np.where(dt > 0, dist_m / dt, 0.0)
        gps_headings = (np.degrees(np.arctan2(pdx, pdy)) + 360.0) % 360.0
        if n_obs > 1: gps_headings[0] = gps_headings[1]
        
        # Robust Adaptive Ground Truth Alignment
        ref_arcs = []
        if route_df is not None and len(route_df) > 0:
            ref_route_arcs = route_df['arc_id'].values
            R = len(ref_route_arcs)
            ru, rv = arcs_df.iloc[ref_route_arcs]['u'].values, arcs_df.iloc[ref_route_arcs]['v'].values
            rx1, ry1 = node_x[ru], node_y[ru]
            rx2, ry2 = node_x[rv], node_y[rv]
            rdx, rdy = rx2 - rx1, ry2 - ry1
            rL2 = rdx*rdx + rdy*rdy
            
            cur_r = 0
            for i in range(n_obs):
                px, py = tx[i], ty[i]
                w_start = max(0, cur_r - 20)
                w_end = min(R, cur_r + 60)
                best_d, best_idx = float('inf'), cur_r
                for r_i in range(w_start, w_end):
                    dx, dy = rdx[r_i], rdy[r_i]
                    L2 = rL2[r_i]
                    t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                    d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                    if d < best_d:
                        best_d, best_idx = d, r_i
                # Fallback to full route if window distance is excessively large
                if best_d > 45.0:
                    for r_i in range(R):
                        dx, dy = rdx[r_i], rdy[r_i]
                        L2 = rL2[r_i]
                        t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                        d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                        if d < best_d:
                            best_d, best_idx = d, r_i
                cur_r = best_idx
                ref_arcs.append(ref_route_arcs[best_idx])
        else:
            ref_arcs = [-1] * n_obs
            
        # Candidate Extraction + Ranking Features
        prev_cand_arcs = set()
        prev_cand_nodes_v = set()
        
        for i in range(n_obs):
            px, py = tx[i], ty[i]
            cx1, cx2 = int((px - radius_m)//cell_size), int((px + radius_m)//cell_size)
            cy1, cy2 = int((py - radius_m)//cell_size), int((py + radius_m)//cell_size)
            cands = set()
            for cx in range(cx1, cx2+1):
                for cy in range(cy1, cy2+1):
                    if (cx, cy) in grid: cands.update(grid[(cx, cy)])
                    
            raw_cands = []
            for aid in cands:
                x1_val, y1_val = ax1[aid], ay1[aid]
                x2_val, y2_val = ax2[aid], ay2[aid]
                dx, dy = x2_val - x1_val, y2_val - y1_val
                L2 = dx*dx + dy*dy
                if L2 == 0:
                    t_val = 0.0
                    d_val = np.hypot(px - x1_val, py - y1_val)
                else:
                    t_val = max(0.0, min(1.0, ((px - x1_val)*dx + (py - y1_val)*dy) / L2))
                    d_val = np.hypot(px - (x1_val + t_val*dx), py - (y1_val + t_val*dy))
                if d_val <= radius_m:
                    h_diff = abs(gps_headings[i] - arc_headings[aid]) % 360.0
                    if h_diff > 180.0: h_diff = 360.0 - h_diff
                    is_connected = 1 if (aid in prev_cand_arcs or u_idx[aid] in prev_cand_nodes_v or i == 0) else 0
                    is_ref = 1 if (aid == ref_arcs[i]) else 0
                    
                    # Advanced distance to endpoints u and v
                    d_u = np.hypot(px - x1_val, py - y1_val)
                    d_v = np.hypot(px - x2_val, py - y2_val)
                    cos_h = np.cos(np.radians(h_diff))
                    
                    raw_cands.append({
                        'trip_id': trip_id,
                        'obs_idx': i,
                        'candidate_arc_id': aid,
                        'is_reference': is_ref,
                        'partition': partition,
                        'dist_to_candidate_arc_m': d_val,
                        'proj_ratio': t_val,
                        'arc_length_m': arc_lengths[aid],
                        'arc_heading_deg': arc_headings[aid],
                        'heading_delta_deg': h_diff,
                        'connected_from_prev': is_connected,
                        'gps_speed_kmh': speed_ms[i] * 3.6,
                        'gps_time_diff_s': dt[i],
                        'gps_dist_m': dist_m[i],
                        'flag_speed_anomaly': 1 if speed_ms[i] > 50.0 else 0,
                        'flag_gap_gt_5s': 1 if dt[i] > 5.0 else 0,
                        'cos_heading_diff': cos_h,
                        'dist_to_u_m': d_u,
                        'dist_to_v_m': d_v
                    })
                    
            if len(raw_cands) > 0:
                # Add relative ranking features within this observation
                raw_cands.sort(key=lambda c: c['dist_to_candidate_arc_m'])
                min_dist = raw_cands[0]['dist_to_candidate_arc_m']
                for rank_idx, c in enumerate(raw_cands):
                    c['dist_rank'] = rank_idx + 1
                    c['dist_ratio_to_min'] = c['dist_to_candidate_arc_m'] / (min_dist + 0.1)
                    c['dist_diff_from_min'] = c['dist_to_candidate_arc_m'] - min_dist
                    
            prev_cand_arcs = set(c['candidate_arc_id'] for c in raw_cands)
            prev_cand_nodes_v = set(v_idx[c['candidate_arc_id']] for c in raw_cands)
            all_rows.extend(raw_cands)
            
    print(f"Generated {len(all_rows):,} enriched candidates for {partition} in {time.time()-t0:.2f}s")
    return pd.DataFrame(all_rows)

    T = len(obs_groups)
    
    # Ground Truth Reference Arcs
    ref_route_arcs = route_df['arc_id'].values
    R = len(ref_route_arcs)
    lat0 = np.mean(nodes_df['lat'])
    m_per_deg_lat = 111139.0
    m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
    node_x = nodes_df['lon'].values * m_per_deg_lon
    node_y = nodes_df['lat'].values * m_per_deg_lat
    tx = track_df['lon'].values * m_per_deg_lon
    ty = track_df['lat'].values * m_per_deg_lat
    ru, rv = arcs_df.iloc[ref_route_arcs]['u'].values, arcs_df.iloc[ref_route_arcs]['v'].values
    rx1, ry1 = node_x[ru], node_y[ru]
    rx2, ry2 = node_x[rv], node_y[rv]
    rdx, rdy = rx2 - rx1, ry2 - ry1
    rL2 = rdx*rdx + rdy*rdy
    
    cur_r = 0
    true_obs_arcs = []
    for i in range(T):
        px, py = tx[i], ty[i]
        w_start = max(0, cur_r - 20)
        w_end = min(R, cur_r + 60)
        best_d, best_idx = float('inf'), cur_r
        for r_i in range(w_start, w_end):
            dx, dy = rdx[r_i], rdy[r_i]
            L2 = rL2[r_i]
            t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
            d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
            if d < best_d:
                best_d, best_idx = d, r_i
        if best_d > 45.0:
            for r_i in range(R):
                dx, dy = rdx[r_i], rdy[r_i]
                L2 = rL2[r_i]
                t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                if d < best_d:
                    best_d, best_idx = d, r_i
        cur_r = best_idx
        true_obs_arcs.append(ref_route_arcs[best_idx])
    true_obs_arcs = np.array(true_obs_arcs)
    
    # Viterbi DP Decoder
    def run_viterbi(prob_col):
        eps = 1e-6
        first_cands = obs_groups[0]['candidate_arc_id'].values
        first_probs = obs_groups[0][prob_col].values
        first_dists = obs_groups[0]['dist_to_candidate_arc_m'].values
        first_emiss = np.log(first_probs + eps) - 0.5 * (first_dists / 15.0)**2
        
        dp = {arc: (first_emiss[i], [arc]) for i, arc in enumerate(first_cands)}
        
        for t in range(1, T):
            curr_cands = obs_groups[t]['candidate_arc_id'].values
            curr_probs = obs_groups[t][prob_col].values
            curr_dists = obs_groups[t]['dist_to_candidate_arc_m'].values
            curr_emiss = np.log(curr_probs + eps) - 0.5 * (curr_dists / 15.0)**2
            
            new_dp = {}
            for c_idx, curr_arc in enumerate(curr_cands):
                e_prob = curr_emiss[c_idx]
                best_prev_score = -float('inf')
                best_path = None
                curr_u = arcs_df.iloc[curr_arc]['u']
                
                for prev_arc, (prev_score, path) in dp.items():
                    prev_v = arcs_df.iloc[prev_arc]['v']
                    if prev_arc == curr_arc:
                        trans_cost = 0.0
                    elif prev_v == curr_u:
                        trans_cost = -0.3
                    else:
                        trans_cost = -4.0
                    total_s = prev_score + trans_cost + e_prob
                    if total_s > best_prev_score:
                        best_prev_score = total_s
                        best_path = path + [curr_arc]
                if best_path is not None:
                    new_dp[curr_arc] = (best_prev_score, best_path)
            dp = new_dp if new_dp else {arc: (curr_emiss[i], [arc]) for i, arc in enumerate(curr_cands)}
            
        best_final_arc = max(dp.keys(), key=lambda k: dp[k][0])
        viterbi_route = np.array(dp[best_final_arc][1])
        
        # Classifier top1
        clf_top1 = np.array([g.loc[g[prob_col].idxmax(), 'candidate_arc_id'] for g in obs_groups])
        
        acc_clf = np.mean(clf_top1 == true_obs_arcs)
        acc_vit = np.mean(viterbi_route == true_obs_arcs)
        
        cont = 0
        for i in range(len(viterbi_route)-1):
            a1, a2 = viterbi_route[i], viterbi_route[i+1]
            if a1 == a2 or arcs_df.iloc[a1]['v'] == arcs_df.iloc[a2]['u']:
                cont += 1
        continuity = cont / (len(viterbi_route)-1) if len(viterbi_route) > 1 else 1.0
        
        return acc_clf, acc_vit, continuity
        
    clf_rf, vit_rf, cont_rf = run_viterbi('prob_rf')
    clf_bl, vit_bl, cont_bl = run_viterbi('prob_blend')
    
    return {
        'trip_id': trip_id,
        'n_obs': T,
        'rf_clf_acc': round(clf_rf * 100, 2),
        'rf_hmm_acc': round(vit_rf * 100, 2),
        'rf_continuity': round(cont_rf * 100, 2),
        'blended_clf_acc': round(clf_bl * 100, 2),
        'blended_hmm_acc': round(vit_bl * 100, 2),
        'blended_continuity': round(cont_bl * 100, 2)
    }

results = []
for tid in test_trips:

    min_samples_leaf = trial.suggest_int('min_samples_leaf', 4, 16, step=2)

    rf = RandomForestClassifier(

        max_depth=max_depth,
        min_samples_leaf=min_samples_leaf,
        min_samples_split=min_samples_split,

                t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                if d < best_d:
                    best_d, best_idx = d, r_i
        cur_r = best_idx
        true_obs_arcs.append(ref_route_arcs[best_idx])
    true_obs_arcs = np.array(true_obs_arcs)
    
    clf_acc = np.mean(clf_top1 == true_obs_arcs)
    vit_acc = np.mean(viterbi_route == true_obs_arcs)
    
    cont = 0
    for i in range(len(viterbi_route)-1):

            cont += 1
    continuity = cont / (len(viterbi_route)-1) if len(viterbi_route) > 1 else 1.0
    
    return {
        'trip_id': trip_id,
        'n_obs': T,
        'clf_acc': clf_acc,
        'vit_acc': vit_acc,
        'continuity': continuity
    }

"""
road_network_hmm_decoder.py
===========================
PathSnap ML Hackathon - Team 11
Graph Theory, HMM & Viterbi Decoder Module

Features:
1. Builds formal directed road network graphs using `networkx.DiGraph`.
2. Computes emission probabilities (ML scores + Gaussian spatial decay).
3. Computes continuous transition probability matrices (Newson & Krumm topology routing).
4. Handles GPS gaps (dt > 5s) and bridges disconnected transitions via shortest paths.
5. Implements log-space Viterbi Dynamic Programming decoding.
6. Benchmarks Classifier-Only vs. Classifier+HMM Route Continuity.
"""

def build_road_graph(nodes_df, arcs_df, lat0):
    """
    Constructs a directed weighted road network graph using networkx.DiGraph.
    Edges are weighted by their physical geodesic length in meters.
    """
    G = nx.DiGraph()
    m_per_deg_lat = 111139.0
    m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
    
    # Add nodes with metric coordinates
    for idx, row in nodes_df.iterrows():
        x = row['lon'] * m_per_deg_lon
        y = row['lat'] * m_per_deg_lat
        G.add_node(idx, x=x, y=y, lon=row['lon'], lat=row['lat'])
        
    # Add arcs as directed edges with arc_id and physical length
    for aid, row in arcs_df.iterrows():
        u = int(row['u'])
        v = int(row['v'])
        if G.has_node(u) and G.has_node(v):
            ux, uy = G.nodes[u]['x'], G.nodes[u]['y']
            vx, vy = G.nodes[v]['x'], G.nodes[v]['y']
            length_m = np.hypot(vx - ux, vy - uy)
            G.add_edge(u, v, arc_id=aid, weight=length_m)
            
    return G

class NetworkHMMViterbiDecoder:
    """
    HMM Viterbi Decoder with Graph-Theoretic Transition Matrices
    and Gaussian Spatial Emission Modeling.
    """
    def __init__(self, graph, arcs_df, sigma_gps=15.0, beta_trans=10.0):
        self.G = graph
        self.arcs_df = arcs_df
        self.sigma_gps = sigma_gps
        self.beta_trans = beta_trans
        self._shortest_path_cache = {}

    def get_network_distance(self, v_start, u_end, cutoff_m=500.0):
        """Computes shortest network distance between two vertices."""
        if v_start == u_end:
            return 0.0
        cache_key = (v_start, u_end)
        if cache_key in self._shortest_path_cache:
            return self._shortest_path_cache[cache_key]
        try:
            d_net = nx.shortest_path_length(self.G, source=v_start, target=u_end, weight='weight')
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            d_net = float('inf')
        self._shortest_path_cache[cache_key] = d_net
        return d_net

    def decode(self, obs_candidates_df):
        """
        Executes Viterbi dynamic programming over GPS observation sequence.
        """
        obs_groups = [g for _, g in obs_candidates_df.groupby('obs_idx')]
        T = len(obs_groups)
        if T == 0:
            return []

        eps = 1e-6
        # Initial step
        first_group = obs_groups[0]
        first_arcs = first_group['candidate_arc_id'].values
        first_probs = first_group['prob'].values
        first_dists = first_group['dist_to_candidate_arc_m'].values
        first_emiss = np.log(first_probs + eps) - 0.5 * (first_dists / self.sigma_gps)**2

        dp = {arc: (first_emiss[i], [arc]) for i, arc in enumerate(first_arcs)}

        for t in range(1, T):
            curr_group = obs_groups[t]
            curr_arcs = curr_group['candidate_arc_id'].values
            curr_probs = curr_group['prob'].values
            curr_dists = curr_group['dist_to_candidate_arc_m'].values
            gps_dt = curr_group['gps_time_diff_s'].values[0]
            gps_disp = curr_group['gps_dist_m'].values[0]
            
            # Spatial Emission Probability
            curr_emiss = np.log(curr_probs + eps) - 0.5 * (curr_dists / self.sigma_gps)**2
            new_dp = {}

            for c_idx, curr_arc in enumerate(curr_arcs):
                curr_u = self.arcs_df.iloc[curr_arc]['u']
                e_prob = curr_emiss[c_idx]
                best_prev_score = -float('inf')
                best_path = None

                for prev_arc, (prev_score, path) in dp.items():
                    prev_v = self.arcs_df.iloc[prev_arc]['v']

                    if prev_arc == curr_arc:
                        # Moving along the same arc
                        trans_log = 0.0
                    elif prev_v == curr_u:
                        # Direct connected topological transition
                        trans_log = -0.2
                    else:
                        # Disconnected / Gap transition: use network graph distance
                        d_net = self.get_network_distance(prev_v, curr_u)
                        if np.isneginf(d_net) or np.isposinf(d_net):
                            # Topological barrier / unreachable in graph
                            trans_log = -5.0
                        else:
                            # Newson & Krumm network-to-Euclidean differential
                            diff = abs(d_net - gps_disp)
                            trans_log = -0.5 - (diff / self.beta_trans)

                    total_score = prev_score + trans_log + e_prob
                    if total_score > best_prev_score:
                        best_prev_score = total_score
                        best_path = path + [curr_arc]

                if best_path is not None:
                    new_dp[curr_arc] = (best_prev_score, best_path)

            dp = new_dp if new_dp else {arc: (curr_emiss[i], [arc]) for i, arc in enumerate(curr_arcs)}

        best_final_arc = max(dp.keys(), key=lambda k: dp[k][0])
        return dp[best_final_arc][1]

def calculate_continuity(arcs_list, arcs_df):
    """Calculates route continuity percentage."""
    if len(arcs_list) <= 1:
        return 1.0
    valid_transitions = 0
    for i in range(len(arcs_list) - 1):
        a1, a2 = arcs_list[i], arcs_list[i+1]
        if a1 == a2 or arcs_df.iloc[a1]['v'] == arcs_df.iloc[a2]['u']:
            valid_transitions += 1
    return (valid_transitions / (len(arcs_list) - 1)) * 100.0

import pandas as pd
import numpy as np


def haversine_distance(lat1, lon1, lat2, lon2):
    """
    Calculate distance between two GPS coordinates in meters.
    """

    earth_radius = 6371000

    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)

    delta_lat = np.radians(lat2 - lat1)
    delta_lon = np.radians(lon2 - lon1)

    a = (
        np.sin(delta_lat / 2) ** 2
        + np.cos(lat1)
        * np.cos(lat2)
        * np.sin(delta_lon / 2) ** 2
    )

    c = 2 * np.arctan2(np.sqrt(a), np.sqrt(1 - a))

    return earth_radius * c


def calculate_bearing(lat1, lon1, lat2, lon2):
    """
    Calculate compass bearing between two GPS points.
    """

    lat1 = np.radians(lat1)
    lat2 = np.radians(lat2)

    delta_lon = np.radians(lon2 - lon1)

    y = np.sin(delta_lon) * np.cos(lat2)

    x = (
        np.cos(lat1) * np.sin(lat2)
        - np.sin(lat1)
        * np.cos(lat2)
        * np.cos(delta_lon)
    )

    bearing = np.degrees(np.arctan2(y, x))

    return (bearing + 360) % 360


def process_gps_trajectory(df):
    """
    Validate and extract basic trajectory features.
    """

    required_columns = [
        "longitude",
        "latitude",
        "timestamp"
    ]

    for column in required_columns:
        if column not in df.columns:
            raise ValueError(
                f"Missing required column: {column}"
            )

    df = df.copy()

    df["longitude"] = pd.to_numeric(
        df["longitude"],
        errors="coerce"
    )

    df["latitude"] = pd.to_numeric(
        df["latitude"],
        errors="coerce"
    )

    df["timestamp"] = pd.to_numeric(
        df["timestamp"],
        errors="coerce"
    )

    df = df.dropna(
        subset=["longitude", "latitude", "timestamp"]
    )

    df = df.reset_index(drop=True)

    if len(df) < 2:
        raise ValueError(
            "Trajectory must contain at least two GPS observations."
        )

    if not df["longitude"].between(-180, 180).all():
        raise ValueError("Invalid longitude values detected.")

    if not df["latitude"].between(-90, 90).all():
        raise ValueError("Invalid latitude values detected.")

    df["time_diff_s"] = (
        df["timestamp"].diff().fillna(0)
    )

    df["dist_m"] = haversine_distance(
        df["latitude"].shift(1),
        df["longitude"].shift(1),
        df["latitude"],
        df["longitude"]
    ).fillna(0)

    df["speed_ms"] = np.where(
        df["time_diff_s"] > 0,
        df["dist_m"] / df["time_diff_s"],
        0
    )

    df["speed_kmh"] = df["speed_ms"] * 3.6

    df["heading_deg"] = calculate_bearing(
        df["latitude"].shift(1),
        df["longitude"].shift(1),
        df["latitude"],
        df["longitude"]
    )

    df["heading_deg"] = df["heading_deg"].fillna(0)

    df["flag_speed_anomaly"] = (
        df["speed_ms"] > 50
    )

    df["flag_gap_gt_5s"] = (
        df["time_diff_s"] > 5
    )

    df["flag_extreme_gap"] = (
        df["time_diff_s"] > 60
    )

    return df

    'dist_to_u_m',
    'dist_to_v_m'
]

def build_enriched_candidates(partition='train', radius_m=50.0):
    all_rows = []
    base_dir = f'docs/Student folder/{partition}'
    trip_dirs = sorted([d for d in glob.glob(os.path.join(base_dir, '*')) if os.path.isdir(d)])
    
    t0 = time.time()
    for td in trip_dirs:
        trip_id = os.path.basename(td)
        track_path = f'{td}/{trip_id}.track'
        nodes_path = f'{td}/{trip_id}.nodes'
        arcs_path = f'{td}/{trip_id}.arcs'
        route_path = f'{td}/{trip_id}.route'
        
        if not (os.path.isfile(track_path) and os.path.isfile(nodes_path) and os.path.isfile(arcs_path)):
            continue
            
        track_df = pd.read_csv(track_path, sep=r'\s+', header=None, names=['lon', 'lat', 'time'])
        nodes_df = pd.read_csv(nodes_path, sep=r'\s+', header=None, names=['lon', 'lat'])
        arcs_df = pd.read_csv(arcs_path, sep=r'\s+', header=None, names=['u', 'v'])
        route_df = pd.read_csv(route_path, header=None, names=['arc_id']) if os.path.isfile(route_path) else None
        
        n_obs = len(track_df)
        lat0 = np.mean(nodes_df['lat'])
        m_per_deg_lat = 111139.0
        m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
        
        node_x = nodes_df['lon'].values * m_per_deg_lon
        node_y = nodes_df['lat'].values * m_per_deg_lat
        
        u_idx, v_idx = arcs_df['u'].values, arcs_df['v'].values
        ax1, ay1 = node_x[u_idx], node_y[u_idx]
        ax2, ay2 = node_x[v_idx], node_y[v_idx]
        adx, ady = ax2 - ax1, ay2 - ay1
        arc_lengths = np.hypot(adx, ady)
        arc_headings = (np.degrees(np.arctan2(adx, ady)) + 360.0) % 360.0
        
        # Spatial Grid Index
        cell_size = float(radius_m)
        grid = {}
        min_x, max_x = np.minimum(ax1, ax2), np.maximum(ax1, ax2)
        min_y, max_y = np.minimum(ay1, ay2), np.maximum(ay1, ay2)
        for aid in range(len(arcs_df)):
            cx1, cx2 = int(min_x[aid]//cell_size), int(max_x[aid]//cell_size)
            cy1, cy2 = int(min_y[aid]//cell_size), int(max_y[aid]//cell_size)
            for cx in range(cx1, cx2+1):
                for cy in range(cy1, cy2+1):
                    grid.setdefault((cx, cy), []).append(aid)
                    
        tx = track_df['lon'].values * m_per_deg_lon
        ty = track_df['lat'].values * m_per_deg_lat
        ttime = track_df['time'].values
        dt = np.zeros(n_obs)
        dt[1:] = ttime[1:] - ttime[:-1]
        pdx, pdy = np.zeros(n_obs), np.zeros(n_obs)
        pdx[1:] = tx[1:] - tx[:-1]
        pdy[1:] = ty[1:] - ty[:-1]
        dist_m = np.hypot(pdx, pdy)
        with np.errstate(divide='ignore', invalid='ignore'):
            speed_ms = np.where(dt > 0, dist_m / dt, 0.0)
        gps_headings = (np.degrees(np.arctan2(pdx, pdy)) + 360.0) % 360.0
        if n_obs > 1: gps_headings[0] = gps_headings[1]
        
        # Robust Adaptive Ground Truth Alignment
        ref_arcs = []
        if route_df is not None and len(route_df) > 0:
            ref_route_arcs = route_df['arc_id'].values
            R = len(ref_route_arcs)
            ru, rv = arcs_df.iloc[ref_route_arcs]['u'].values, arcs_df.iloc[ref_route_arcs]['v'].values
            rx1, ry1 = node_x[ru], node_y[ru]
            rx2, ry2 = node_x[rv], node_y[rv]
            rdx, rdy = rx2 - rx1, ry2 - ry1
            rL2 = rdx*rdx + rdy*rdy
            
            cur_r = 0
            for i in range(n_obs):
                px, py = tx[i], ty[i]
                w_start = max(0, cur_r - 20)
                w_end = min(R, cur_r + 60)
                best_d, best_idx = float('inf'), cur_r
                for r_i in range(w_start, w_end):
                    dx, dy = rdx[r_i], rdy[r_i]
                    L2 = rL2[r_i]
                    t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                    d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                    if d < best_d:
                        best_d, best_idx = d, r_i
                # Fallback to full route if window distance is excessively large
                if best_d > 45.0:
                    for r_i in range(R):
                        dx, dy = rdx[r_i], rdy[r_i]
                        L2 = rL2[r_i]
                        t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                        d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                        if d < best_d:
                            best_d, best_idx = d, r_i
                cur_r = best_idx
                ref_arcs.append(ref_route_arcs[best_idx])
        else:
            ref_arcs = [-1] * n_obs
            
        # Candidate Extraction + Ranking Features
        prev_cand_arcs = set()
        prev_cand_nodes_v = set()
        
        for i in range(n_obs):
            px, py = tx[i], ty[i]
            cx1, cx2 = int((px - radius_m)//cell_size), int((px + radius_m)//cell_size)
            cy1, cy2 = int((py - radius_m)//cell_size), int((py + radius_m)//cell_size)
            cands = set()
            for cx in range(cx1, cx2+1):
                for cy in range(cy1, cy2+1):
                    if (cx, cy) in grid: cands.update(grid[(cx, cy)])
                    
            raw_cands = []
            for aid in cands:
                x1_val, y1_val = ax1[aid], ay1[aid]
                x2_val, y2_val = ax2[aid], ay2[aid]
                dx, dy = x2_val - x1_val, y2_val - y1_val
                L2 = dx*dx + dy*dy
                if L2 == 0:
                    t_val = 0.0
                    d_val = np.hypot(px - x1_val, py - y1_val)
                else:
                    t_val = max(0.0, min(1.0, ((px - x1_val)*dx + (py - y1_val)*dy) / L2))
                    d_val = np.hypot(px - (x1_val + t_val*dx), py - (y1_val + t_val*dy))
                if d_val <= radius_m:
                    h_diff = abs(gps_headings[i] - arc_headings[aid]) % 360.0
                    if h_diff > 180.0: h_diff = 360.0 - h_diff
                    is_connected = 1 if (aid in prev_cand_arcs or u_idx[aid] in prev_cand_nodes_v or i == 0) else 0
                    is_ref = 1 if (aid == ref_arcs[i]) else 0
                    
                    # Advanced distance to endpoints u and v
                    d_u = np.hypot(px - x1_val, py - y1_val)
                    d_v = np.hypot(px - x2_val, py - y2_val)
                    cos_h = np.cos(np.radians(h_diff))
                    
                    raw_cands.append({
                        'trip_id': trip_id,
                        'obs_idx': i,
                        'candidate_arc_id': aid,
                        'is_reference': is_ref,
                        'partition': partition,
                        'dist_to_candidate_arc_m': d_val,
                        'proj_ratio': t_val,
                        'arc_length_m': arc_lengths[aid],
                        'arc_heading_deg': arc_headings[aid],
                        'heading_delta_deg': h_diff,
                        'connected_from_prev': is_connected,
                        'gps_speed_kmh': speed_ms[i] * 3.6,
                        'gps_time_diff_s': dt[i],
                        'gps_dist_m': dist_m[i],
                        'flag_speed_anomaly': 1 if speed_ms[i] > 50.0 else 0,
                        'flag_gap_gt_5s': 1 if dt[i] > 5.0 else 0,
                        'cos_heading_diff': cos_h,
                        'dist_to_u_m': d_u,
                        'dist_to_v_m': d_v
                    })
                    
            if len(raw_cands) > 0:
                # Add relative ranking features within this observation
                raw_cands.sort(key=lambda c: c['dist_to_candidate_arc_m'])
                min_dist = raw_cands[0]['dist_to_candidate_arc_m']
                for rank_idx, c in enumerate(raw_cands):
                    c['dist_rank'] = rank_idx + 1
                    c['dist_ratio_to_min'] = c['dist_to_candidate_arc_m'] / (min_dist + 0.1)
                    c['dist_diff_from_min'] = c['dist_to_candidate_arc_m'] - min_dist
                    
            prev_cand_arcs = set(c['candidate_arc_id'] for c in raw_cands)
            prev_cand_nodes_v = set(v_idx[c['candidate_arc_id']] for c in raw_cands)
            all_rows.extend(raw_cands)
            
    print(f"Generated {len(all_rows):,} enriched candidates for {partition} in {time.time()-t0:.2f}s")
    return pd.DataFrame(all_rows)

def evaluate_trip_models(trip_id):
    trip_cands = test_df[test_df['trip_id'] == trip_id].copy()
    if len(trip_cands) == 0:
        return None
        
    td = f'docs/Student folder/student_test/{trip_id}'
    nodes_df = pd.read_csv(f'{td}/{trip_id}.nodes', sep=r'\s+', header=None, names=['lon', 'lat'])
    arcs_df = pd.read_csv(f'{td}/{trip_id}.arcs', sep=r'\s+', header=None, names=['u', 'v'])
    route_df = pd.read_csv(f'{td}/{trip_id}.route', header=None, names=['arc_id'])
    track_df = pd.read_csv(f'{td}/{trip_id}.track', sep=r'\s+', header=None, names=['lon', 'lat', 'time'])
    
    # Model predictions
    probs_rf = rf_advanced.predict_proba(trip_cands[ADVANCED_FEATURE_COLS].values)[:, 1]
    probs_hgb = hgb_advanced.predict_proba(trip_cands[ADVANCED_FEATURE_COLS].values)[:, 1]
    # Blended probability
    probs_blend = 0.60 * probs_rf + 0.40 * probs_hgb
    
    trip_cands['prob_rf'] = probs_rf
    trip_cands['prob_blend'] = probs_blend
    
    obs_groups = [group for _, group in trip_cands.groupby('obs_idx')]
    T = len(obs_groups)
    
    # Ground Truth Reference Arcs
    ref_route_arcs = route_df['arc_id'].values
    R = len(ref_route_arcs)
    lat0 = np.mean(nodes_df['lat'])
    m_per_deg_lat = 111139.0
    m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
    node_x = nodes_df['lon'].values * m_per_deg_lon
    node_y = nodes_df['lat'].values * m_per_deg_lat
    tx = track_df['lon'].values * m_per_deg_lon
    ty = track_df['lat'].values * m_per_deg_lat
    ru, rv = arcs_df.iloc[ref_route_arcs]['u'].values, arcs_df.iloc[ref_route_arcs]['v'].values
    rx1, ry1 = node_x[ru], node_y[ru]
    rx2, ry2 = node_x[rv], node_y[rv]
    rdx, rdy = rx2 - rx1, ry2 - ry1
    rL2 = rdx*rdx + rdy*rdy
    
    cur_r = 0
    true_obs_arcs = []
    for i in range(T):
        px, py = tx[i], ty[i]
        w_start = max(0, cur_r - 20)
        w_end = min(R, cur_r + 60)
        best_d, best_idx = float('inf'), cur_r
        for r_i in range(w_start, w_end):
            dx, dy = rdx[r_i], rdy[r_i]
            L2 = rL2[r_i]
            t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
            d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
            if d < best_d:
                best_d, best_idx = d, r_i
        if best_d > 45.0:
            for r_i in range(R):
                dx, dy = rdx[r_i], rdy[r_i]
                L2 = rL2[r_i]
                t_val = 0.0 if L2==0 else max(0.0, min(1.0, ((px - rx1[r_i])*dx + (py - ry1[r_i])*dy)/L2))
                d = np.hypot(px - (rx1[r_i] + t_val*dx), py - (ry1[r_i] + t_val*dy))
                if d < best_d:
                    best_d, best_idx = d, r_i
        cur_r = best_idx
        true_obs_arcs.append(ref_route_arcs[best_idx])
    true_obs_arcs = np.array(true_obs_arcs)
    
    # Viterbi DP Decoder
    def run_viterbi(prob_col):
        eps = 1e-6
        first_cands = obs_groups[0]['candidate_arc_id'].values
        first_probs = obs_groups[0][prob_col].values
        first_dists = obs_groups[0]['dist_to_candidate_arc_m'].values
        first_emiss = np.log(first_probs + eps) - 0.5 * (first_dists / 15.0)**2
        
        dp = {arc: (first_emiss[i], [arc]) for i, arc in enumerate(first_cands)}
        
        for t in range(1, T):
            curr_cands = obs_groups[t]['candidate_arc_id'].values
            curr_probs = obs_groups[t][prob_col].values
            curr_dists = obs_groups[t]['dist_to_candidate_arc_m'].values
            curr_emiss = np.log(curr_probs + eps) - 0.5 * (curr_dists / 15.0)**2
            
            new_dp = {}
            for c_idx, curr_arc in enumerate(curr_cands):
                e_prob = curr_emiss[c_idx]
                best_prev_score = -float('inf')
                best_path = None
                curr_u = arcs_df.iloc[curr_arc]['u']
                
                for prev_arc, (prev_score, path) in dp.items():
                    prev_v = arcs_df.iloc[prev_arc]['v']
                    if prev_arc == curr_arc:
                        trans_cost = 0.0
                    elif prev_v == curr_u:
                        trans_cost = -0.3
                    else:
                        trans_cost = -4.0
                    total_s = prev_score + trans_cost + e_prob
                    if total_s > best_prev_score:
                        best_prev_score = total_s
                        best_path = path + [curr_arc]
                if best_path is not None:
                    new_dp[curr_arc] = (best_prev_score, best_path)
            dp = new_dp if new_dp else {arc: (curr_emiss[i], [arc]) for i, arc in enumerate(curr_cands)}
            
        best_final_arc = max(dp.keys(), key=lambda k: dp[k][0])
        viterbi_route = np.array(dp[best_final_arc][1])
        
        # Classifier top1
        clf_top1 = np.array([g.loc[g[prob_col].idxmax(), 'candidate_arc_id'] for g in obs_groups])
        
        acc_clf = np.mean(clf_top1 == true_obs_arcs)
        acc_vit = np.mean(viterbi_route == true_obs_arcs)
        
        cont = 0
        for i in range(len(viterbi_route)-1):
            a1, a2 = viterbi_route[i], viterbi_route[i+1]
            if a1 == a2 or arcs_df.iloc[a1]['v'] == arcs_df.iloc[a2]['u']:
                cont += 1
        continuity = cont / (len(viterbi_route)-1) if len(viterbi_route) > 1 else 1.0
        
        return acc_clf, acc_vit, continuity
        
    clf_rf, vit_rf, cont_rf = run_viterbi('prob_rf')
    clf_bl, vit_bl, cont_bl = run_viterbi('prob_blend')
    
    return {
        'trip_id': trip_id,
        'n_obs': T,
        'rf_clf_acc': round(clf_rf * 100, 2),
        'rf_hmm_acc': round(vit_rf * 100, 2),
        'rf_continuity': round(cont_rf * 100, 2),
        'blended_clf_acc': round(clf_bl * 100, 2),
        'blended_hmm_acc': round(vit_bl * 100, 2),
        'blended_continuity': round(cont_bl * 100, 2)
    }

results = []
for tid in test_trips:
    res = evaluate_trip_models(tid)
    results.append(res)

eval_df = pd.DataFrame(results)

total_obs = eval_df['n_obs'].sum()

avg_rf_hmm = np.average(eval_df['rf_hmm_acc'], weights=eval_df['n_obs'])
avg_rf_cont = np.average(eval_df['rf_continuity'], weights=eval_df['n_obs'])

avg_bl_clf = np.average(eval_df['blended_clf_acc'], weights=eval_df['n_obs'])
avg_bl_hmm = np.average(eval_df['blended_hmm_acc'], weights=eval_df['n_obs'])
avg_bl_cont = np.average(eval_df['blended_continuity'], weights=eval_df['n_obs'])

"""
road_network_hmm_decoder.py
===========================
PathSnap ML Hackathon - Team 11
Graph Theory, HMM & Viterbi Decoder Module

Features:
1. Builds formal directed road network graphs using `networkx.DiGraph`.
2. Computes emission probabilities (ML scores + Gaussian spatial decay).
3. Computes continuous transition probability matrices (Newson & Krumm topology routing).
4. Handles GPS gaps (dt > 5s) and bridges disconnected transitions via shortest paths.
5. Implements log-space Viterbi Dynamic Programming decoding.
6. Benchmarks Classifier-Only vs. Classifier+HMM Route Continuity.
"""

import time
import os
import joblib
import networkx as nx
import numpy as np
import pandas as pd

ADVANCED_FEATURE_COLS = [
    'dist_to_candidate_arc_m', 'proj_ratio', 'arc_length_m', 'arc_heading_deg',
    'heading_delta_deg', 'connected_from_prev', 'gps_speed_kmh', 'gps_dist_m',
    'gps_time_diff_s', 'flag_speed_anomaly', 'flag_gap_gt_5s',
    'cos_heading_diff', 'dist_rank', 'dist_ratio_to_min', 'dist_diff_from_min',
    'dist_to_u_m', 'dist_to_v_m'
]

def build_road_graph(nodes_df, arcs_df, lat0):
    """
    Constructs a directed weighted road network graph using networkx.DiGraph.
    Edges are weighted by their physical geodesic length in meters.
    """
    G = nx.DiGraph()
    m_per_deg_lat = 111139.0
    m_per_deg_lon = 111139.0 * np.cos(np.radians(lat0))
    
    # Add nodes with metric coordinates
    for idx, row in nodes_df.iterrows():
        x = row['lon'] * m_per_deg_lon
        y = row['lat'] * m_per_deg_lat
        G.add_node(idx, x=x, y=y, lon=row['lon'], lat=row['lat'])
        
    # Add arcs as directed edges with arc_id and physical length
    for aid, row in arcs_df.iterrows():
        u = int(row['u'])
        v = int(row['v'])
        if G.has_node(u) and G.has_node(v):
            ux, uy = G.nodes[u]['x'], G.nodes[u]['y']
            vx, vy = G.nodes[v]['x'], G.nodes[v]['y']
            length_m = np.hypot(vx - ux, vy - uy)
            G.add_edge(u, v, arc_id=aid, weight=length_m)
            
    return G

class NetworkHMMViterbiDecoder:
    """
    HMM Viterbi Decoder with Graph-Theoretic Transition Matrices
    and Gaussian Spatial Emission Modeling.
    """
    def __init__(self, graph, arcs_df, sigma_gps=15.0, beta_trans=10.0):
        self.G = graph
        self.arcs_df = arcs_df
        self.sigma_gps = sigma_gps
        self.beta_trans = beta_trans
        self._shortest_path_cache = {}

    def get_network_distance(self, v_start, u_end, cutoff_m=500.0):
        """Computes shortest network distance between two vertices."""
        if v_start == u_end:
            return 0.0
        cache_key = (v_start, u_end)
        if cache_key in self._shortest_path_cache:
            return self._shortest_path_cache[cache_key]
        try:
            d_net = nx.shortest_path_length(self.G, source=v_start, target=u_end, weight='weight')
        except (nx.NetworkXNoPath, nx.NodeNotFound):
            d_net = float('inf')
        self._shortest_path_cache[cache_key] = d_net
        return d_net

    def decode(self, obs_candidates_df):
        """
        Executes Viterbi dynamic programming over GPS observation sequence.
        """
        obs_groups = [g for _, g in obs_candidates_df.groupby('obs_idx')]
        T = len(obs_groups)
        if T == 0:
            return []

        eps = 1e-6
        # Initial step
        first_group = obs_groups[0]
        first_arcs = first_group['candidate_arc_id'].values
        first_probs = first_group['prob'].values
        first_dists = first_group['dist_to_candidate_arc_m'].values
        first_emiss = np.log(first_probs + eps) - 0.5 * (first_dists / self.sigma_gps)**2

        dp = {arc: (first_emiss[i], [arc]) for i, arc in enumerate(first_arcs)}

        for t in range(1, T):
            curr_group = obs_groups[t]
            curr_arcs = curr_group['candidate_arc_id'].values
            curr_probs = curr_group['prob'].values
            curr_dists = curr_group['dist_to_candidate_arc_m'].values
            gps_dt = curr_group['gps_time_diff_s'].values[0]
            gps_disp = curr_group['gps_dist_m'].values[0]
            
            # Spatial Emission Probability
            curr_emiss = np.log(curr_probs + eps) - 0.5 * (curr_dists / self.sigma_gps)**2
            new_dp = {}

            for c_idx, curr_arc in enumerate(curr_arcs):
                curr_u = self.arcs_df.iloc[curr_arc]['u']
                e_prob = curr_emiss[c_idx]
                best_prev_score = -float('inf')
                best_path = None

                for prev_arc, (prev_score, path) in dp.items():
                    prev_v = self.arcs_df.iloc[prev_arc]['v']

                    if prev_arc == curr_arc:
                        # Moving along the same arc
                        trans_log = 0.0
                    elif prev_v == curr_u:
                        # Direct connected topological transition
                        trans_log = -0.2
                    else:
                        # Disconnected / Gap transition: use network graph distance
                        d_net = self.get_network_distance(prev_v, curr_u)
                        if np.isneginf(d_net) or np.isposinf(d_net):
                            # Topological barrier / unreachable in graph
                            trans_log = -5.0
                        else:
                            # Newson & Krumm network-to-Euclidean differential
                            diff = abs(d_net - gps_disp)
                            trans_log = -0.5 - (diff / self.beta_trans)

                    total_score = prev_score + trans_log + e_prob
                    if total_score > best_prev_score:
                        best_prev_score = total_score
                        best_path = path + [curr_arc]

                if best_path is not None:
                    new_dp[curr_arc] = (best_prev_score, best_path)

            dp = new_dp if new_dp else {arc: (curr_emiss[i], [arc]) for i, arc in enumerate(curr_arcs)}

        best_final_arc = max(dp.keys(), key=lambda k: dp[k][0])
        return dp[best_final_arc][1]

def calculate_continuity(arcs_list, arcs_df):
    """Calculates route continuity percentage."""
    if len(arcs_list) <= 1:
        return 1.0
    valid_transitions = 0
    for i in range(len(arcs_list) - 1):
        a1, a2 = arcs_list[i], arcs_list[i+1]
        if a1 == a2 or arcs_df.iloc[a1]['v'] == arcs_df.iloc[a2]['u']:
            valid_transitions += 1
    return (valid_transitions / (len(arcs_list) - 1)) * 100.0
